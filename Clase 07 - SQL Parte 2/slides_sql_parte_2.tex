\begin{frame}{Cláusula HAVING}
    \begin{center}
        {\Large \faFilter}\\[0.2cm]
        \textbf{Filtrar resultados de grupos}
    \end{center}
    
    \vspace{0.3cm}
    
    \textbf{Diferencia con WHERE:}
    \begin{itemize}
        \item[{\makebox[1em][c]{\textcolor{black}{\faTable}}}] \textbf{WHERE}: filtra registros individuales
        \item[{\makebox[1em][c]{\textcolor{black}{\faUsers}}}] \textbf{HAVING}: filtra grupos después de GROUP BY
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Ejemplo de HAVING}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
SELECT departamento, COUNT(*) as cantidad_empleados
FROM empleados
GROUP BY departamento
HAVING COUNT(*) > 2;
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Ejemplo de HAVING con resultados}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Departamentos con mas de 2 empleados
SELECT departamento, COUNT(*) as cantidad_empleados
FROM empleados
GROUP BY departamento
HAVING COUNT(*) > 2;
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{cc}
            \toprule
            departamento & cantidad\_empleados \\
            \midrule
            Ventas & 3 \\
            IT & 4 \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}{Subconsultas}
    \begin{center}
        {\Large \faCode}\\[0.2cm]
        \textbf{Consultas anidadas dentro de otras consultas}
    \end{center}
    
    \vspace{0.3cm}
    
    \textbf{Tipos de subconsultas:}
    \begin{itemize}
        \item[{\makebox[1em][c]{\textcolor{black}{\faFilter}}}] \textbf{En WHERE}: filtrar con resultados de otra consulta
        \item[{\makebox[1em][c]{\textcolor{black}{\faTable}}}] \textbf{En FROM}: usar como tabla temporal
        \item[{\makebox[1em][c]{\textcolor{black}{\faCheck}}}] \textbf{Con EXISTS}: verificar existencia
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Subconsulta en WHERE}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Empleados con salario mayor al promedio
SELECT nombre, salario
FROM empleados
WHERE salario > (
    SELECT AVG(salario)
    FROM empleados
);
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{cc}
            \toprule
            nombre & salario \\
            \midrule
            María & 60000 \\
            Juan & 55000 \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}[fragile]{Subconsulta en FROM}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Departamentos con promedio de salario > 50000
SELECT dept_info.departamento, 
       dept_info.promedio_salario
FROM (
    SELECT departamento, 
           AVG(salario) as promedio_salario
    FROM empleados
    GROUP BY departamento
) as dept_info
WHERE dept_info.promedio_salario > 50000;
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{cc}
            \toprule
            departamento & promedio\_salario \\
            \midrule
            IT & 55000 \\
            Ventas & 52000 \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}[fragile]{Subconsulta con EXISTS}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Empleados que han hecho ventas > 10000
SELECT nombre
FROM empleados e
WHERE EXISTS (
    SELECT 1 
    FROM ventas v
    WHERE v.id_empleado = e.id 
    AND v.monto > 10000
);
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{c}
            \toprule
            nombre \\
            \midrule
            Juan \\
            María \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}{Funciones de Ventana}
    \begin{center}
        \textbf{Cálculos a través de conjuntos de filas relacionadas}
    \end{center}
    
    \vspace{0.3cm}
    
    \textbf{Características:}
    \begin{itemize}
        \item No agrupan filas como GROUP BY
        \item Mantienen todas las filas originales
        \item Agregan columnas calculadas
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{RANK()}
    \begin{center}
        {\Large \faTrophy}\\[0.2cm]
        \textbf{Asignar rangos a filas}
    \end{center}
    
    \vspace{0.3cm}
    
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]

            Juan & 55000 & 2 \\
            Carlos & 55000 & 2 \\
            Pedro & 45000 & 4 \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}[fragile]{PARTITION BY}
    \begin{center}
        \textbf{Dividir en grupos para cálculos}
    \end{center}
    
    \vspace{0.3cm}
    
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Promedio de salario por departamento
SELECT 
    nombre,
    departamento,
    salario,
    AVG(salario) OVER (
        PARTITION BY departamento
    ) as promedio_dept
FROM empleados;
    \end{lstlisting}
    
    \vspace{0.3cm}

        \begin{tabular}{cccc}

            Carlos & Ventas & 55000 & 50000 \\
            Pedro & Ventas & 45000 & 50000 \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}[fragile]{ROW\_NUMBER()}

\begin{frame}{Common Table Expressions (CTE)}
    \begin{center}
        \textbf{Subconsultas mas legibles y reutilizables}
    \end{center}
    
    \vspace{0.3cm}
    
    \textbf{Ventajas:}
    \begin{itemize}
        \item Mejor legibilidad
        \item Reutilizacion de logica
        \item Simplifica consultas complejas
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Ejemplo de CTE}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
WITH salarios_dept AS (
    SELECT 
        departamento,
        AVG(salario) as promedio_salario
    FROM empleados
    GROUP BY departamento
)
SELECT e.nombre,
       e.salario,
       s.promedio_salario
FROM empleados e
JOIN salarios_dept s 
    ON e.departamento = s.departamento
WHERE e.salario > s.promedio_salario;
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{ccc}
            \toprule
            nombre & salario & promedio\_salario \\
            \midrule
            María & 60000 & 55000 \\
            Carlos & 55000 & 50000 \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}
