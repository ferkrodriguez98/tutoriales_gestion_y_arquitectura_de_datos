\begin{frame}[fragile]{Estructura básica de una consulta}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
SELECT columna1, columna2
FROM tabla
WHERE condicion;
    \end{lstlisting}
    
    \vspace{0.5cm}
    
    \textbf{Componentes principales:}
    \begin{itemize}
        \item[{\makebox[1em][c]{\textcolor{black}{\faEye}}}] \textbf{SELECT}: especifica qué columnas queremos ver
        \item[{\makebox[1em][c]{\textcolor{black}{\faTable}}}] \textbf{FROM}: indica de qué tabla obtenemos los datos
        \item[{\makebox[1em][c]{\textcolor{black}{\faFilter}}}] \textbf{WHERE}: filtra los resultados (opcional)
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Ejemplo de consulta básica}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Buscar empleados con salario mayor a 50000
SELECT nombre, apellido, salario
FROM empleados
WHERE salario > 50000;
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{ccc}
            \toprule
            nombre & apellido & salario \\
            \midrule
            Juan & Pérez & 55000 \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

                \textbf{AND}\\[0.1cm]

    \begin{itemize}

\begin{frame}[fragile]{Operador IN}
    \begin{center}
        {\Large \faList}\\[0.2cm]
        \textbf{Buscar valores en una lista}
    \end{center}
    
    \vspace{0.3cm}
    
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Buscar empleados con apellidos especificos
SELECT nombre, apellido
FROM empleados
WHERE apellido IN ('Garcia', 'Lopez');
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{cc}
            \toprule
            nombre & apellido \\
            \midrule
            Juan & García \\
            María & López \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}[fragile]{Operador BETWEEN}
    \begin{center}
        {\Large \faArrowsH}\\[0.2cm]
        \textbf{Buscar valores entre dos rangos}
    \end{center}
    
    \vspace{0.3cm}
    
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Buscar empleados con salario entre 40000 y 50000
SELECT nombre, apellido, salario

    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{ccc}
            \toprule

        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}[fragile]{Operador LIKE}
    \begin{center}
        {\Large \faSearch}\\[0.2cm]
        \textbf{Buscar patrones en texto}
    \end{center}
    
    \vspace{0.3cm}
    
    \textbf{Comodines:}
    \begin{itemize}
        \item \textbf{\%} cualquier cantidad de caracteres
        \item \textbf{\_} un único caracter
    \end{itemize}
    
    \vspace{0.3cm}
    
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Apellidos que terminan en G
SELECT nombre, apellido
FROM empleados
WHERE apellido LIKE '%G';

-- Apellidos que empiezan con G
SELECT nombre, apellido
FROM empleados
WHERE apellido LIKE 'G%';
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Valores NULL}
    \begin{center}
        {\Large \faQuestionCircle}\\[0.2cm]
        \textbf{Representa la ausencia de un valor}
    \end{center}
    
    \vspace{0.3cm}
    
    \textbf{Operadores para NULL:}
    \begin{itemize}
        \item[{\makebox[1em][c]{\textcolor{black}{\faCheck}}}] \textbf{IS NULL}: verificar si es NULL
        \item[{\makebox[1em][c]{\textcolor{black}{\faTimes}}}] \textbf{IS NOT NULL}: verificar si NO es NULL
    \end{itemize}
    
    \vspace{0.3cm}
    
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Buscar empleados sin telefono
SELECT nombre, telefono
FROM empleados
WHERE telefono IS NULL;
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{cc}
            \toprule
            nombre & telefono \\
            \midrule
            Carlos & NULL \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}


\begin{frame}[fragile]{ORDER BY}
    \begin{center}
        \textbf{Ordenar resultados}
    \end{center}
    
    \textbf{Opciones de ordenamiento:}
    \begin{itemize}
        \item[{\makebox[1em][c]{\textcolor{black}{\faArrowUp}}}] \textbf{ASC}: ascendente (por defecto)
        \item[{\makebox[1em][c]{\textcolor{black}{\faArrowDown}}}] \textbf{DESC}: descendente
    \end{itemize}

    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Ordenar por edad descendente
SELECT nombre, apellido, edad
FROM empleados
ORDER BY edad DESC;
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{ccc}
            \toprule
            nombre & apellido & edad \\
            \midrule
            Carlos & López & 45 \\
            María & García & 35 \\
            Juan & Pérez & 30 \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}[fragile]{Alias}
    \begin{center}
        {\Large \faTag}\\[0.2cm]
        \textbf{Nombres temporales para tablas o columnas}
    \end{center}
    
    \vspace{0.3cm}
    
    \textbf{Tipos de alias:}
    \begin{itemize}
        \item[{\makebox[1em][c]{\textcolor{black}{\faTable}}}] \textbf{Alias de tablas}: \texttt{empleados AS e}
        \item[{\makebox[1em][c]{\textcolor{black}{\faColumns}}}] \textbf{Alias de columnas}: \texttt{nombre AS nombre\_empleado}
    \end{itemize}
    
    \vspace{0.3cm}
    
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Usando alias
SELECT 
    e.nombre AS nombre_empleado,
    e.salario AS sueldo_mensual
FROM empleados e
WHERE e.salario > 50000;
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Ejemplo de funciones de agregación}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Estadisticas de empleados
SELECT 
    COUNT(*) as total_empleados,
    SUM(salario) as total_salarios,
    AVG(salario) as salario_promedio,
    MAX(salario) as salario_maximo,
    MIN(salario) as salario_minimo
FROM empleados;
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
    \resizebox{\textwidth}{!}{
    \begin{tabular}{ccccc}
        \toprule
        total\_empleados & total\_salarios & salario\_promedio & salario\_maximo & salario\_minimo \\
        \midrule
        3 & 155000 & 51666.67 & 60000 & 45000 \\
        \bottomrule
    \end{tabular}
    }
    \end{center}

\begin{frame}[fragile]{GROUP BY}
    \begin{center}
        {\Large \faUsers}\\[0.2cm]

\begin{frame}[fragile]{Ejemplo de INNER JOIN}
    \begin{lstlisting}[language=SQL, numbers=left, numbersep=5pt]
-- Combinar empleados con departamentos
SELECT e.nombre, d.nombre_departamento
FROM empleados e
INNER JOIN departamentos d ON e.id_departamento = d.id;
    \end{lstlisting}
    
    \vspace{0.3cm}
    
    \begin{center}
        \begin{tabular}{cc}
            \toprule
            nombre & nombre\_departamento \\
            \midrule
            Juan & Ventas \\
            María & IT \\
            Carlos & Ventas \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}[fragile]{Ejemplo de LEFT JOIN}

-- Todos los empleados, con departamento si existe

LEFT JOIN departamentos d ON e.id_departamento = d.id;
